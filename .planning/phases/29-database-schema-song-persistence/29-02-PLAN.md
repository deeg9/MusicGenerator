```typescript
export const songStatusEnum = pgEnum('song_status', ['generating', 'completed', 'failed']);

export const songs = pgTable('songs', {
  id: uuid('id').defaultRandom().primaryKey(),
  userId: uuid('user_id'),
  sessionToken: text('session_token'),
  status: songStatusEnum('status').notNull().default('generating'),
  heartbulaJobId: text('heartbula_job_id').notNull(),
  recipientName: text('recipient_name').notNull(),
  relationship: text('relationship').notNull(),
  genre: text('genre').notNull(),
  quizData: jsonb('quiz_data').notNull().$type<{...}>(),
  r2WavKey: text('r2_wav_key'),
  r2Mp3Key: text('r2_mp3_key'),
  createdAt: timestamp('created_at', { withTimezone: true }).defaultNow().notNull(),
  completedAt: timestamp('completed_at', { withTimezone: true }),
  failedAt: timestamp('failed_at', { withTimezone: true }),
}, ...).enableRLS();
```
