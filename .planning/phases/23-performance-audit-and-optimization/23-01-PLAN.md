**In src/app/quiz/success/[jobId]/_components/ConfettiBlast.tsx:**

Remove the top-level `import confetti from 'canvas-confetti'` and replace with dynamic `import()` inside the useEffect. Use the cleanup-safe pattern from research:

```typescript
'use client';

import { useEffect } from 'react';
import { useReducedMotion } from '@/hooks/useReducedMotion';

const BRAND_COLORS = ['#4a90d9', '#f5d060', '#ffffff', '#7bb8f5'];

export function ConfettiBlast() {
  const reducedMotion = useReducedMotion();

  useEffect(() => {
    if (reducedMotion) return;
    let timer: ReturnType<typeof setTimeout>;

    void import('canvas-confetti').then(({ default: confetti }) => {
      void confetti({
        origin: { x: 0.25, y: 0.5 },
        particleCount: 80,
        spread: 70,
        colors: BRAND_COLORS,
      });
      timer = setTimeout(() => {
        void confetti({
          origin: { x: 0.75, y: 0.5 },
          particleCount: 80,
          spread: 70,
          colors: BRAND_COLORS,
        });
      }, 150);
    });

    return () => clearTimeout(timer);
  }, [reducedMotion]);

  return null;
}
```

Critical details:
- Destructure `{ default: confetti }` from the dynamic import (ESM/CJS interop)
- Declare `timer` with `let` before the `.then()` so cleanup can access it
- The cleanup `clearTimeout(timer)` works because if the component unmounts before `.then()` resolves, `timer` is undefined and clearTimeout(undefined) is a no-op
